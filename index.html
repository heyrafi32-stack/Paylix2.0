} function money(amount) { return "৳" + Number(amount || 0).toFixed(2); } function makeId() { return Date.now().toString(36) + Math.random().toString(36).slice(2, 7); } /* REGISTER */ function registerUser(name, phone, password) { if (!name || !phone || !password) { alert("সব তথ্য পূরণ করুন"); return; } const user = { id: makeId(), name, phone, password, balance: 0, referralCode: "PAY" + Math.random().toString(36).slice(2, 8).toUpperCase(), blocked: false, createdAt: new Date().toLocaleString() }; users.push(user); saveData(DB.users, users); alert("রেজিস্ট্রেশন সফল হয়েছে"); } /* LOGIN */ function loginUser(phone, password) { const users = getData(DB.users); const user = users.find( u => u.phone === phone && u.password === password ); } /* LOGOUT */ function logoutUser() { localStorage.removeItem(DB.current); location.href = "index.html"; } /* USER BALANCE */ function updateBalance(userId, amount) { const users = getData(DB.users); if (loggedIn && loggedIn.id === userId) { localStorage.setItem(DB.current, JSON.stringify(users[index])); } } /* GMAIL TASK SUBMISSION */ function submitGmailTask(email, taskCode) { const user = currentUser(); } /* ADMIN APPROVE GMAIL TASK */ function approveGmailTask(submissionId) { const submissions = getData(DB.gmail); const index = submissions.findIndex(s => s.id === submissionId); if (index === -1 || submissions[index].status !== "Pending") return; saveData(DB.tasks, tasks); updateBalance(submission.userId, task.reward); alert("Task Approved। User-এর Balance-এ টাকা যোগ হয়েছে।"); } /* ADMIN REJECT MICRO TASK */ function rejectMicroTask(taskId, submissionId) { const tasks = getData(DB.tasks); const task = tasks.find(t => t.id === taskId); if (!task) return; const submission = task.submissions.find(s => s.id === submissionId); if (!submission || submission.status !== "Pending") return; submission.status = "Rejected"; saveData(DB.tasks, tasks); alert("Task Rejected"); } /* ADMIN DELETE MICRO TASK */ function deleteMicroTask(taskId) { const tasks = getData(DB.tasks); saveData(DB.tasks, tasks.filter(t => t.id !== taskId)); if (index === -1 || users[index].balance < amount) { alert("আপনার Balance পর্যাপ্ত নয়"); return; } users[index].balance -= amount; saveData(DB.users, users); localStorage.setItem(DB.current, JSON.stringify(users[index])); const withdrawals = getData(DB.withdrawals); withdrawals.push({ id: makeId(), userId: user.id, name: user.name, phone: user.phone, amount, paymentNumber, method: "bKash", status: "Pending", date: new Date().toLocaleString() }); saveData(DB.withdrawals, withdrawals); alert("Withdrawal Request জমা হয়েছে"); } /* ADMIN APPROVE WITHDRAWAL */ function approveWithdrawal(requestId) { const requests = getData(DB.withdrawals); const index = requests.findIndex(r => r.id === requestId); if (index === -1 || requests[index].status !== "Pending") return; requests[index].status = "Approved"; saveData(DB.withdrawals, requests); alert("Withdrawal Approved"); } /* ADMIN REJECT WITHDRAWAL */ function rejectWithdrawal(requestId) { const requests = getData(DB.withdrawals);
